📦 Pacote sugerido: com.seuprojeto.application.annotation

package com.seuprojeto.application.annotation;

import java.lang.annotation.*;

/**
 * Indica que a classe ou método deve ser monitorado e enviado ao Dashboard.
 *
 * <p>Se nenhum campo for informado, o corpo completo da requisição e resposta será enviado.</p>
 *
 * <p>Para limitar os dados enviados, informe quais propriedades do corpo devem ser incluídas usando os
 * campos {@code includedRequestFields} e {@code includedResponseFields}.</p>
 *
 * <pre>
 * Exemplos:
 *   @MonitoredEndpoint
 *   → Envia tudo.
 *
 *   @MonitoredEndpoint(includedRequestFields = {"email", "cpf"})
 *   → Envia apenas os campos "email" e "cpf" do request.
 * </pre>
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.TYPE, ElementType.METHOD})
public @interface MonitoredEndpoint {

    /**
     * Campos específicos do corpo da requisição que devem ser enviados.
     * Se vazio, envia todo o corpo da requisição.
     */
    String[] includedRequestFields() default {};

    /**
     * Campos específicos do corpo da resposta que devem ser enviados.
     * Se vazio, envia todo o corpo da resposta.
     */
    String[] includedResponseFields() default {};
}


✅ ETAPA 2: Criar PayloadFilter para aplicar os filtros
📦 Pacote sugerido: com.seuprojeto.application.util

package com.seupacote.filtro;

@Component
@RequiredArgsConstructor
public class PayloadFilter {

    private final ObjectMapper objectMapper;

    public Map<String, Object> filterPayload(String json, List<String> allowedFields) {
        if (json == null || json.isEmpty() || allowedFields == null || allowedFields.isEmpty()) {
            return Collections.emptyMap();
        }

        try {
            JsonNode root = objectMapper.readTree(json);
            Map<String, Object> result = new LinkedHashMap<>();

            for (String path : allowedFields) {
                addFilteredField(result, root, path);
            }

            return result;
        } catch (Exception e) {
            return Collections.emptyMap();
        }
    }

    private void addFilteredField(Map<String, Object> result, JsonNode currentNode, String path) {
        String[] parts = path.split("\\.");
        addRecursive(result, currentNode, parts, 0);
    }

    @SuppressWarnings("unchecked")
    private void addRecursive(Map<String, Object> result, JsonNode node, String[] path, int index) {
        String key = path[index];
        JsonNode child = node.get(key);

        if (child == null || child.isNull()) return;

        if (index == path.length - 1) {
            // Último nível, adiciona valor simples
            result.putIfAbsent(key, child.isValueNode() ? child.asText() : convertNode(child));
        } else {
            if (child.isArray()) {
                // Se for array, aplica recursivamente em cada item
                List<Object> filteredList = new ArrayList<>();
                for (JsonNode item : child) {
                    Map<String, Object> filteredItem = new LinkedHashMap<>();
                    addRecursive(filteredItem, item, path, index + 1);
                    filteredList.add(filteredItem);
                }
                result.putIfAbsent(key, filteredList);
            } else if (child.isObject()) {
                Map<String, Object> nested = (Map<String, Object>) result.getOrDefault(key, new LinkedHashMap<>());
                addRecursive(nested, child, path, index + 1);
                result.put(key, nested);
            }
        }
    }

    private Object convertNode(JsonNode node) {
        try {
            return objectMapper.treeToValue(node, Object.class);
        } catch (Exception e) {
            return null;
        }
    }
}


ETAPA 3: Atualizar MonitoredEndpointRegistry
📦 Pacote atual: com.seuprojeto.application.registry

package com.seuprojeto.application.registry;

import com.seuprojeto.application.annotation.MonitoredEndpoint;
import jakarta.annotation.PostConstruct;
import lombok.Getter;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.ApplicationContext;
import org.springframework.core.annotation.AnnotatedElementUtils;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.servlet.mvc.method.annotation.RequestMappingHandlerMapping;

import java.lang.reflect.Method;
import java.util.*;

@Slf4j
@Component
public class MonitoredEndpointRegistry {

    private final ApplicationContext applicationContext;

    // Armazena nome do controller + método -> true/false se é monitorado
    private final Set<String> monitoredEndpoints = new HashSet<>();

    // Armazena os campos configurados para filtro por método
    private final Map<String, List<String>> requestFieldsMap = new HashMap<>();
    private final Map<String, List<String>> responseFieldsMap = new HashMap<>();

    public MonitoredEndpointRegistry(ApplicationContext applicationContext) {
        this.applicationContext = applicationContext;
    }

    @PostConstruct
    public void init() {
        log.info("Inicializando MonitoredEndpointRegistry...");

        Map<String, RequestMappingHandlerMapping> mappings = applicationContext.getBeansOfType(RequestMappingHandlerMapping.class);
        for (RequestMappingHandlerMapping mapping : mappings.values()) {
            Map<RequestMappingInfo, HandlerMethod> handlerMethods = mapping.getHandlerMethods();

            for (HandlerMethod method : handlerMethods.values()) {
                Class<?> beanType = method.getBeanType();
                Method methodObj = method.getMethod();

                MonitoredEndpoint classAnnotation = AnnotatedElementUtils.findMergedAnnotation(beanType, MonitoredEndpoint.class);
                MonitoredEndpoint methodAnnotation = AnnotatedElementUtils.findMergedAnnotation(methodObj, MonitoredEndpoint.class);

                if (classAnnotation != null || methodAnnotation != null) {
                    String key = buildKey(beanType.getSimpleName(), methodObj.getName());
                    monitoredEndpoints.add(key);

                    // Preferência para a anotação do método se ambos existirem
                    MonitoredEndpoint effectiveAnnotation = methodAnnotation != null ? methodAnnotation : classAnnotation;

                    requestFieldsMap.put(key, Arrays.asList(effectiveAnnotation.includeRequestFields()));
                    responseFieldsMap.put(key, Arrays.asList(effectiveAnnotation.includeResponseFields()));

                    log.info("Registrado endpoint monitorado: {} com filtros req={}, res={}",
                            key,
                            Arrays.toString(effectiveAnnotation.includeRequestFields()),
                            Arrays.toString(effectiveAnnotation.includeResponseFields()));
                }
            }
        }
    }

    private String buildKey(String controllerName, String methodName) {
        return controllerName + "#" + methodName;
    }

    public boolean isMonitored(String controllerName, String methodName) {
        return monitoredEndpoints.contains(buildKey(controllerName, methodName));
    }

    public List<String> getIncludedRequestFields(String controllerName, String methodName) {
        return requestFieldsMap.getOrDefault(buildKey(controllerName, methodName), Collections.emptyList());
    }

    public List<String> getIncludedResponseFields(String controllerName, String methodName) {
        return responseFieldsMap.getOrDefault(buildKey(controllerName, methodName), Collections.emptyList());
    }
}


Por que usar Map<String, MonitoredEndpoint> para métodos e classes?
Com isso, você consegue:

Saber exatamente de onde a anotação veio (classe ou método).

Ter acesso à configuração dos campos incluídos/excluídos diretamente pela annotation.

package com.seuprojeto.application.interceptor;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.seuprojeto.application.registry.MonitoredEndpointRegistry;
import com.seuprojeto.domain.enums.AwsSqsQueueEnum;
import com.seuprojeto.domain.model.DashboardData;
import com.seuprojeto.domain.model.DashboardMessage;
import com.seuprojeto.domain.port.MessagePublisherPort;
import com.seuprojeto.infrastructure.filter.PayloadFilter;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.servlet.HandlerInterceptor;
import org.springframework.web.util.ContentCachingRequestWrapper;
import org.springframework.web.util.ContentCachingResponseWrapper;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;
import java.util.*;

@Slf4j
@Component
public class DashboardInterceptor implements HandlerInterceptor {

    private final MessagePublisherPort messagePublisher;
    private final MonitoredEndpointRegistry endpointRegistry;
    private final ObjectMapper objectMapper;
    private final PayloadFilter payloadFilter;

    public DashboardInterceptor(
            MessagePublisherPort messagePublisher,
            MonitoredEndpointRegistry endpointRegistry,
            ObjectMapper objectMapper,
            PayloadFilter payloadFilter
    ) {
        this.messagePublisher = messagePublisher;
        this.endpointRegistry = endpointRegistry;
        this.objectMapper = objectMapper;
        this.payloadFilter = payloadFilter;
    }

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) throws Exception {
        if (!(handler instanceof HandlerMethod handlerMethod)) {
            return;
        }

        String controllerName = handlerMethod.getBeanType().getSimpleName();
        String methodName = handlerMethod.getMethod().getName();

        if (!endpointRegistry.isMonitored(controllerName, methodName)) {
            return;
        }

        // Pega os headers
        Map<String, String> headers = new HashMap<>();
        Enumeration<String> headerNames = request.getHeaderNames();
        while (headerNames.hasMoreElements()) {
            String headerName = headerNames.nextElement();
            headers.put(headerName, request.getHeader(headerName));
        }

        // Leitura original dos bodies
        String requestBody = getRequestBody(request);
        String responseBody = getResponseBody(response);

        // Pega os campos filtrados definidos na anotação
        List<String> includedRequestFields = endpointRegistry.getIncludedRequestFields(controllerName, methodName);
        List<String> includedResponseFields = endpointRegistry.getIncludedResponseFields(controllerName, methodName);

        // Aplica filtro se necessário
        if (!includedRequestFields.isEmpty()) {
            Map<String, Object> filteredRequest = payloadFilter.filterPayload(requestBody, includedRequestFields);
            requestBody = objectMapper.writeValueAsString(filteredRequest);
        }

        if (!includedResponseFields.isEmpty()) {
            Map<String, Object> filteredResponse = payloadFilter.filterPayload(responseBody, includedResponseFields);
            responseBody = objectMapper.writeValueAsString(filteredResponse);
        }

        // Monta o payload final
        DashboardData dashboardData = new DashboardData(
                request.getRequestURI(),
                LocalDateTime.now(),
                headers,
                requestBody,
                responseBody
        );

        String payload = objectMapper.writeValueAsString(dashboardData);
        log.info("Payload enviado ao dashboard: {}", payload);

        DashboardMessage message = DashboardMessage.fromDashboardData(dashboardData);
        messagePublisher.publishToQueue(message, AwsSqsQueueEnum.METRICS_SQS_TAILOR);
    }

    private String getRequestBody(HttpServletRequest request) {
        if (request instanceof ContentCachingRequestWrapper cachingRequest) {
            byte[] content = cachingRequest.getContentAsByteArray();
            return content.length > 0 ? new String(content, StandardCharsets.UTF_8) : "";
        }
        return "";
    }

    private String getResponseBody(HttpServletResponse response) throws IOException {
        if (response instanceof ContentCachingResponseWrapper cachingResponse) {
            byte[] content = cachingResponse.getContentAsByteArray();
            String responseBody = content.length > 0 ? new String(content, StandardCharsets.UTF_8) : "";
            cachingResponse.copyBodyToResponse();
            return responseBody;
        }
        return "";
    }
}

package com.seuprojeto.application.interceptor;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.node.ObjectNode;
import com.seuprojeto.domain.model.DashboardData;
import com.seuprojeto.domain.model.DashboardMessage;
import com.seuprojeto.application.annotation.MonitoredEndpoint;
import com.seuprojeto.application.registry.MonitoredEndpointRegistry;
import com.seuprojeto.domain.port.MessagePublisherPort;
import com.seuprojeto.infrastructure.aws.enums.AwsSqsQueueEnum;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.util.ContentCachingRequestWrapper;
import org.springframework.web.util.ContentCachingResponseWrapper;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;
import java.util.*;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

@Component
public class DashboardInterceptor implements HandlerInterceptor {

    private static final Logger log = LoggerFactory.getLogger(DashboardInterceptor.class);

    private final MessagePublisherPort messagePublisher;
    private final MonitoredEndpointRegistry endpointRegistry;
    private final ObjectMapper objectMapper;

    public DashboardInterceptor(MessagePublisherPort messagePublisher, MonitoredEndpointRegistry endpointRegistry, ObjectMapper objectMapper) {
        this.messagePublisher = messagePublisher;
        this.endpointRegistry = endpointRegistry;
        this.objectMapper = objectMapper;
    }

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) throws Exception {
        if (!(handler instanceof HandlerMethod handlerMethod)) {
            return;
        }

        String controllerName = handlerMethod.getBeanType().getSimpleName();
        String methodName     = handlerMethod.getMethod().getName();

        if (!endpointRegistry.isMonitored(controllerName, methodName)) {
            return;
        }

        Optional<MonitoredEndpoint> annotationOpt = endpointRegistry.getAnnotation(controllerName, methodName);

        List<String> includedRequestFields  = annotationOpt.map(a -> Arrays.asList(a.includedRequestFields())).orElse(Collections.emptyList());
        List<String> includedResponseFields = annotationOpt.map(a -> Arrays.asList(a.includedResponseFields())).orElse(Collections.emptyList());

        String rawRequestBody  = getRequestBody(request);
        String rawResponseBody = getResponseBody(response);

        String filteredRequestBody  = filterJsonFields(rawRequestBody, includedRequestFields);
        String filteredResponseBody = filterJsonFields(rawResponseBody, includedResponseFields);

        Map<String, String> headers = extractHeaders(request);

        DashboardData dashboardData = new DashboardData(
                request.getRequestURI(),
                LocalDateTime.now(),
                headers,
                filteredRequestBody,
                filteredResponseBody
        );

        String payload = objectMapper.writeValueAsString(dashboardData);
        log.info("payload enviado ao dashboard: {}", payload);

        DashboardMessage message = DashboardMessage.fromDashboardData(dashboardData);
        messagePublisher.publishToQueue(message, AwsSqsQueueEnum.METRICS_SQS_TAILOR);
    }

    private String getRequestBody(HttpServletRequest request) {
        if (request instanceof ContentCachingRequestWrapper cachingRequest) {
            byte[] content = cachingRequest.getContentAsByteArray();
            return content.length > 0 ? new String(content, StandardCharsets.UTF_8) : "";
        }
        return "";
    }

    private String getResponseBody(HttpServletResponse response) throws IOException {
        if (response instanceof ContentCachingResponseWrapper cachingResponse) {
            byte[] content = cachingResponse.getContentAsByteArray();
            String responseBody = content.length > 0 ? new String(content, StandardCharsets.UTF_8) : "";
            cachingResponse.copyBodyToResponse(); // importantíssimo
            return responseBody;
        }
        return "";
    }

    private Map<String, String> extractHeaders(HttpServletRequest request) {
        Map<String, String> headers = new HashMap<>();
        Enumeration<String> headerNames = request.getHeaderNames();
        while (headerNames != null && headerNames.hasMoreElements()) {
            String headerName = headerNames.nextElement();
            headers.put(headerName, request.getHeader(headerName));
        }
        return headers;
    }

    private String filterJsonFields(String jsonBody, List<String> includedFields) {
        if (includedFields.isEmpty() || jsonBody == null || jsonBody.isBlank()) {
            return jsonBody;
        }

        try {
            JsonNode rootNode = objectMapper.readTree(jsonBody);
            ObjectNode filteredNode = objectMapper.createObjectNode();

            for (String field : includedFields) {
                if (rootNode.has(field)) {
                    filteredNode.set(field, rootNode.get(field));
                }
            }

            return objectMapper.writeValueAsString(filteredNode);
        } catch (Exception e) {
            log.warn("Erro ao filtrar campos do JSON: {}", e.getMessage());
            return jsonBody;
        }
    }
}
